const {createReadStream} = require('fs')

const stream = createReadStream('./4.txt',{highWaterMark:90000,encoding: 'utf8})

stream.on('data',(result)=>{
    console.log(result);
    
})

stream.on('error',(err)=>{
    console.log(err)
})
// When you call createReadStream('./4.txt'),
//  it creates a readable stream for the file 4.txt.
//  This stream reads the file in chunks instead of loading the entire file into memory.

// The 'data' event is triggered whenever
//  a new chunk of data is available, 
// and Node.js automatically passes that chunk as an 
// argument (result) to the callback function you provided


//default 64kb


-----



let http = require('http')
let fs = require('fs')

http.createServer((req,res) =>{
    const fileStream = fs.createReadStream('./4.txt','utf8')
    fileStream.on('open',()=>{
        fileStream.pipe(res)
    })
    fileStream.on('error',(err)=>{
        res.end(err)
    })
})
.listen(5000)

//fileStream.pipe(res) is using the pipe() method to
//  efficiently transfer data from the file stream 
// (fileStream) to the HTTP response (res).
