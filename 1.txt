can also use setInterval() and setTimeout()
example:
setInterval(()=>{
    console.log('Hello world')
},1000)
This code snippet uses JavaScript's setInterval() function
 to execute a block of code at regular intervals 
 (in milliseconds).


 The above code is an arrow function u can also do the same using this:
 setInterval(function () {
  console.log("Hello World");
}, 2000);

 or

function printMessage() {
  console.log("Hello, this message repeats every 2 seconds!");
}

setInterval(printMessage, 2000);


Now, how to stop?
use setTimeout() :
example 
let intervalId = setInterval(() => {
    console.log('Hello world');
}, 1000);

// Stop the interval after 5 seconds
setTimeout(() => {
    clearInterval(intervalId);
    console.log('Interval cleared');
}, 5000);

so after five seconds it will stop

using es6 modules example:

// file: math.js
export default function multiply(a, b) {
  return a * b;
}

export const PI = 3.14;

//remember If you want to make parts of the main.js file
 accessible to other JavaScript files, 
 you need to explicitly export them. 

// file: main.js
import multiply, { PI } from './math.js';

console.log(multiply(2, 3)); // 6
console.log(PI); // 3.14

To use ES6 modules in the browser:

Use the type="module" attribute in the <script> tag:
<script type="module" src="main.js"></script>

In CommonJS, every file is module by default
Modules - Encapsulated Code (only share minimum)

ex:

//names.js
const john = "john"
const peter = "peter"
const Afzaal = "AFzaal Ahmed Khan"

module.exports = {john,peter}

//function.js
const sayHi = (name) => {
    console.log(`Hello ${name}, what u doing??`);
}

module.exports = sayHi

//app.js
const names = require('./names.js')
const sayHi = require('./function.js')

sayHi(names.john)
sayHi(names.peter)

console.log(names);


or 

const names = require('./names.js')
const ftogreetname = require('./function.js')

//sayHi(Afzaal)
ftogreetname(names.john)
ftogreetname(names.peter)
ftogreetname("Afzaal")

console.log(names);

path module 
Examples:
const path = require('path')

console.log(path.sep);
//returns the platform specific path
//segment separator
//in windows \


const filePath = path.join('content','subfolder','test.txt')
console.log(filePath);
//output : content\subfolder\test.txt

const baseName = path.basename(filePath)
console.log(baseName);
//output : test.txt

const absolute = path.resolve(__dirname,'content','subfolder','test.txt')
console.log(absolute);
//path.resolve() method in nodejs is used to create
//an absolute path by resolving a sequence of path
//segments.
//output : C:\node js\content\subfolder\test.txt

fs module


there are two flavors , we can do it 
async (non-blocking) or sync(blocking)


const {readFileSync , writeFileSync } = require('fs')
//Extracts (destructures) the readFileSync and writeFileSync
//  functions from the fs module.

//readFileSync: Reads files synchronously
//  (blocking execution until the file is read).

//writeFileSync: Writes data to a file synchronously
//  (blocking execution until the writing is complete).


const first = readFileSync("check.txt","utf-8")
console.log(first);

writeFileSync('./check.txt',`Here is the result : ${first}`)

writeFileSync('./check.txt',`Here is the result : ${first}`,{flag: 'a'})



--async
example:
const {readFile , writeFile } = require('fs')


readFile("check.txt",'utf-8',(err,result)=> {
    if(err){
        console.log(err);
        return
    }
    console.log(result);    
})

//readFile("check.txt", "utf-8", callback)

// Reads the file check.txt asynchronously.
// Uses "utf-8" to properly decode text instead of returning a buffer.
//err: Holds an error object if something goes wrong.
//result: The file content (as a string if encoding is provided, otherwise as a Buffer).
// If an error occurs, logs it.
// Otherwise, logs the file contents.



good example:

const {readFile , writeFile } = require('fs')


readFile("check.txt",'utf-8',(err,result)=> {
    if(err){
        console.log(err);
        return
    }
    const first = result
    console.log(first);
    
    readFile("names.js",'utf-8',(err,result)=>{
    if(err){
        console.log(err);
        return
    }
    const second = result
    console.log(second);
    
    writeFile('./check.txt',
        `Here is the result : ${first}     ,     ${second}`,
        (err,result) =>{
            if(err){
                console.log(err);
                return
            }
            console.log(result);  
        }
    )
    })
})
